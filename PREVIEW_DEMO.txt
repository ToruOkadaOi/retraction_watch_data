TESTING YOUR PREVIEW URL
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Once you have a preview URL, test it:

  # Save your URL
  PREVIEW="https://your-preview-url.com"
  
  # Test root endpoint
  curl "$PREVIEW/"
  
  # Query retractions
  curl "$PREVIEW/retractions?year=2025&page_size=5"
  
  # Get specific retraction
  curl "$PREVIEW/retractions/69157"
  
  # List top authors
  curl "$PREVIEW/authors?limit=10"
  
  # View API documentation
  open "$PREVIEW/docs"
